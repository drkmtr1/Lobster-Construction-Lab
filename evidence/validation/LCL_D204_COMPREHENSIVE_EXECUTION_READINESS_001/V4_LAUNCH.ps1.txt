$ErrorActionPreference = 'Stop'
$out = Join-Path (Get-Location) 'evidence/validation/LCL_D204_COMPREHENSIVE_EXECUTION_READINESS_001/V4_STDOUT.txt'
$err = Join-Path (Get-Location) 'evidence/validation/LCL_D204_COMPREHENSIVE_EXECUTION_READINESS_001/V4_STDERR.txt'
if ((Test-Path -LiteralPath $out) -or (Test-Path -LiteralPath $err) -or (Test-Path -LiteralPath 'runs/d204-readiness-v4')) { throw 'Occupied qualification identity' }
$exe = (& .\.venv\Scripts\python.exe -I -S -B -c 'import sys; print(sys._base_executable)')
if ($LASTEXITCODE -ne 0) { exit $LASTEXITCODE }
$script = Join-Path (Get-Location) 'research/d204_readiness_v4/entry.py'
$vector = @('-I','-S','-B',('"' + $script + '"'),'LCL-D204-COMPREHENSIVE-EXECUTION-READINESS-001','qualify','90f86ace9eb2fa78110f66bd9c2aa732553e651562f06b0b9969128f21b88d4b','9d4e438e52d789e108c149e9279baad3b7ac9a7ee522f3957511dcc16fe3d116')
$watch = [Diagnostics.Stopwatch]::StartNew()
$outer = Start-Process -FilePath $exe -ArgumentList $vector -WorkingDirectory (Get-Location) -WindowStyle Hidden -RedirectStandardOutput $out -RedirectStandardError $err -PassThru
$timeout = -not $outer.WaitForExit(790000)
if ($timeout) {
    $outer.Kill()
    $closed = $outer.WaitForExit(5000)
} else {
    $closed = $true
}
$elapsed = $watch.ElapsedMilliseconds
$total = $null
$user = $null
$kernel = $null
$code = $null
if ($closed) {
    $code = $outer.ExitCode
    try {
        $total = $outer.TotalProcessorTime.TotalSeconds
        $user = $outer.UserProcessorTime.TotalSeconds
        $kernel = $outer.PrivilegedProcessorTime.TotalSeconds
    } catch {}
}
[ordered]@{pid=$outer.Id;controller_signaled=$closed;outer_timeout=$timeout;actual_outer_elapsed_ms=$elapsed;exit_code=$code;controller_cpu_total=$total;controller_cpu_user=$user;controller_cpu_kernel=$kernel} | ConvertTo-Json -Compress
