$ErrorActionPreference = 'Stop'
$out = Join-Path (Get-Location) 'evidence/validation/LCL_D204_CONSOLE_DETACHED_EXECUTION_READINESS_002/V7_STDOUT.txt'
$err = Join-Path (Get-Location) 'evidence/validation/LCL_D204_CONSOLE_DETACHED_EXECUTION_READINESS_002/V7_STDERR.txt'
if ((Test-Path -LiteralPath $out) -or (Test-Path -LiteralPath $err) -or (Test-Path -LiteralPath 'runs/d204-readiness-v7')) { throw 'Occupied qualification identity' }
$exe = (& .\.venv\Scripts\python.exe -I -S -B -c 'import sys; print(sys._base_executable)')
if ($LASTEXITCODE -ne 0) { exit $LASTEXITCODE }
$script = Join-Path (Get-Location) 'research/d204_readiness_v7/entry.py'
$vector = @('-I','-S','-B',('"' + $script + '"'),'LCL-D204-CONSOLE-DETACHED-EXECUTION-READINESS-002','qualify','d5b906993db2553a6009f2e6220a9f795539d20c0c6856d9e4d4d78b1a9b6225','a51fa0bbac1ea5750b685d2d5208d75d5d490f98f49010d8853be9de0202a8be')

$outer = Start-Process -FilePath $exe -ArgumentList $vector -WorkingDirectory (Get-Location) -WindowStyle Hidden -RedirectStandardOutput $out -RedirectStandardError $err -PassThru
$timeout = -not $outer.WaitForExit(790000)
if ($timeout) {
    $outer.Kill()
    $closed = $outer.WaitForExit(5000)
} else {
    $closed = $true
}
$elapsed = $watch.ElapsedMilliseconds
$total = $null
$user = $null
$kernel = $null
$code = $null

    $code = $outer.ExitCode
    try {
        $total = $outer.TotalProcessorTime.TotalSeconds
        $user = $outer.UserProcessorTime.TotalSeconds
        $kernel = $outer.PrivilegedProcessorTime.TotalSeconds
    } catch {}
}
