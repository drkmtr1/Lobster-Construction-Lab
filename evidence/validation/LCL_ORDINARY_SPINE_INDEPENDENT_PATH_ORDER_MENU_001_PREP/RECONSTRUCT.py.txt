def reconstruct(profile):
 spine=[f"s:{i}" for i in range(len(profile))]
 vertices=spine.copy()
 roles={name:{"kind":"spine","section":i} for i,name in enumerate(spine)}
 edges=[list(pair) for pair in zip(spine,spine[1:])]
 for i,row in enumerate(profile):
    for j,count in enumerate(row):
        branch=f"b:{i}:{j}"
        vertices.append(branch)
        roles[branch]={"kind":"branch","section":i,"branch":j}
        edges.append([spine[i],branch])
        for k in range(count):
            leaf=f"l:{i}:{j}:{k}"
            vertices.append(leaf)
            roles[leaf]={"kind":"leaf","section":i,"branch":j,"leaf":k}
            edges.append([branch,leaf])
 return dict(schema_version=1,profile=profile,vertices=vertices,edges=edges,spine=spine,roles=roles)

def canonical(graph):
 return dict(profile=graph["profile"],spine=graph["spine"],roles=graph["roles"],
    vertices=sorted(graph["vertices"]),edges=sorted(sorted(e) for e in graph["edges"]))
def digest(graph):
 return hashlib.sha256(json.dumps(canonical(graph),sort_keys=True,separators=(",",":")).encode()).hexdigest()

def table(graph,labels):
 n=len(graph["edges"])
 assert set(labels)==set(graph["vertices"]) and sorted(labels.values())==list(range(n+1))
 ranks=[]
 for a,b in graph["edges"]:
    x,y=sorted([labels[a],labels[b]])
    ranks.append(dict(rank=y-x,first_cut=x+1,last_cut=y,ends=[a,b],endpoint_labels=[x,y]))
 ranks.sort(key=lambda row:row["rank"])
 assert [r["rank"] for r in ranks]==list(range(1,n+1))
 cuts=[]
 for h in range(1,n+1):
    values=[r["rank"] for r in ranks if r["first_cut"]<=h<=r["last_cut"]]
    assert values
    cuts.append(dict(cut=h,crossings=values,count=len(values),safe_threshold=values[0] if values==list(range(values[0],n+1)) else None))
 return ranks,cuts
