import hashlib,json,sys,time
from pathlib import Path
root=Path("evidence/validation/LCL_ORDINARY_SPINE_INDEPENDENT_PATH_ORDER_MENU_001_PREP")
prefixes=json.loads((root/"PREFIXES.json").read_bytes())
original=json.loads((root/"ORIGINALS.json").read_bytes())["records"]["main_41"]["prefixes"][0]["graph"]
installed=set(prefixes["initializer"]["graph"]["vertices"])
fixed=prefixes["shell_fixed_maximum_labels"]
def slots(graph, labels, cut, allocated):
 records={"cut":cut,"named_labels":{},"named_edges":{}}
 for key,names in (("old",installed),("fresh",set(graph["vertices"])-installed)):
    named={name:labels[name] for name in sorted(set(labels)&names)}
    records["named_labels"][key]={side:{name:x for name,x in named.items() if (x<cut)==lower}
                                 for side,lower in (("lower",True),("upper",False))}
 for kind in ("old","fresh"):
    list_rows=[]
    for u,v in graph["edges"]:
        if u not in labels or v not in labels or (kind=="old")!=(u in installed and v in installed): continue
        x,y=labels[u],labels[v]
        list_rows.append(dict(names=[u,v],endpoints=[x,y],weight=abs(x-y),crosses=(x<cut)!=(y<cut)))
    records["named_edges"][kind]=list_rows
    records[kind+"_weights"]={k:sorted(t["weight"] for t in list_rows if t["crosses"]==c)
                               for k,c in (("crossing",True),("noncrossing",False))}
 rows=records["named_edges"]["old"]+records["named_edges"]["fresh"]
 used_labels=set(labels.values())
 used_weights={row["weight"] for row in rows}
 assert len(used_labels)==len(labels) and len(used_weights)==len(rows)
 records["remaining_names"]=sorted(set(graph["vertices"])-set(labels))
 records["remaining_labels"]=sorted(set(range(33))-used_labels)
 records["remaining_weights"]=sorted(set(range(1,33))-used_weights)
 records["connector"]=next(row for row in rows if row["names"]==["s:4","s:5"])
 assert len(records["remaining_labels"])==len(records["remaining_weights"])==len(records["remaining_names"])==32-allocated
 return records
def mirror(labels):return {name:32-x for name,x in labels.items()}
partials={"maximum":slots(original,fixed,13,17),"minimum":slots(original,mirror(fixed),20,17)}
assert partials["maximum"]["remaining_labels"]==prefixes["source_r"]
assert partials["minimum"]["remaining_labels"]==sorted(32-r for r in prefixes["source_r"])
assert partials["maximum"]["remaining_weights"]==partials["minimum"]["remaining_weights"]==prefixes["source_q"]
menus={}
for label,row in prefixes["menu"].items():
 menus[label]={}
 for mode,cut in (("minimum",20),("maximum",13)):
    labels=row["state"][mode+"_labels"]
    receipt=slots(original,labels,cut,32)
    weights=sorted(receipt["old_weights"]["crossing"]+receipt["fresh_weights"]["crossing"])
    assert weights==list(range(12,33))
    menus[label][mode]=receipt
def write(name,obj):
 with (root/name).open("x",encoding="utf8",newline="\n") as f:
    json.dump(obj,f,sort_keys=True,indent=2)
    f.write("\n")
write("INVENTORIES.json",dict(partial=partials,complete=menus))
def ident(path):
 raw=Path(path).read_bytes()
 return {"bytes":len(raw),"sha256":hashlib.sha256(raw).hexdigest()}
exe=Path(sys._base_executable)
assert sys.version_info[:2] in ((3,12),(3,13)) and sys.flags.isolated and sys.flags.no_site and sys.dont_write_bytecode
write("RUNTIME.json",dict(platform=sys.platform,implementation=sys.implementation.name,version=sys.version,
    executable=str(exe),interpreter=ident(exe),dll=ident(exe.with_name("python313.dll")),
    stdlib_only=True,task_imports=0,bytecode_generated=0,observed_process_cpu_seconds=time.process_time(),
    scope="Fixed arithmetic and parse-only preparation, not authorization or qualification of a future scientific launcher."))
